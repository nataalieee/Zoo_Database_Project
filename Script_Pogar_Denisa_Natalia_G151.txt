-- SCRIPT CREATE

--- TABELUL CONTRACT
CREATE TABLE CONTRACT (
    id_contract NUMBER(5),
    salariu NUMBER(8, 2),
    tip_contract VARCHAR2(25) NOT NULL, 
    data_semnarii DATE DEFAULT SYSDATE,
    zile_concediu_disp NUMBER(2),
    constraint pk_contract PRIMARY KEY(id_contract),
    CHECK (salariu>0),
    CHECK (upper(tip_contract) IN ('DETERMINAT', 'NEDETERMINAT', 'PART-TIME'))
);

---TABELUL ANGAJAT
CREATE TABLE ANGAJAT (
    id_angajat NUMBER(5),
    id_contract NUMBER(5),
    nume VARCHAR2(25) NOT NULL,
    prenume VARCHAR2(25) NOT NULL,
    cnp CHAR(13) NOT NULL,
    telefon CHAR(10) NOT NULL,
    email VARCHAR2(100),
    adresa VARCHAR2(100),
    functie VARCHAR2(25),
    data_angajarii DATE,
    constraint pk_angajat PRIMARY KEY(id_angajat),
    constraint fk_contract_angajat FOREIGN KEY (id_contract) REFERENCES CONTRACT(id_contract),
    constraint uk_cnp UNIQUE(cnp),
    constraint uk_telefon UNIQUE(telefon)
);

---Tabel SECTOR
CREATE TABLE SECTOR (
    id_sector NUMBER(5),
    denumire VARCHAR2(50) NOT NULL,
    suprafata NUMBER(8, 2) NOT NULL,
    tip_habitat VARCHAR2(30),
    cap_maxima NUMBER(3), 
    temp_medie NUMBER(4, 1), 
    CONSTRAINT pk_sector PRIMARY KEY (id_sector),
    CONSTRAINT uk_denumire_sector UNIQUE (denumire),
    CONSTRAINT check_suprafata CHECK (suprafata>0),
    CONSTRAINT check_cap_maxima CHECK (cap_maxima>0)
);

---TABEL ANIMAL
CREATE TABLE ANIMAL (
    id_animal NUMBER(5),
    id_sector NUMBER(5), 
    tip_animal VARCHAR2(20) NOT NULL,
    data_nasterii DATE,
    data_venirii DATE DEFAULT SYSDATE,
    specie VARCHAR2(50) NOT NULL,
    sex CHAR(1),
    status_sanatate VARCHAR2(50),
    CONSTRAINT pk_animal PRIMARY KEY (id_animal),
    CONSTRAINT fk_animal_sector FOREIGN KEY (id_sector) REFERENCES SECTOR(id_sector),
    CONSTRAINT check_date_animal CHECK (data_venirii >= data_nasterii),
    CONSTRAINT check_sex CHECK (UPPER(sex) IN ('M', 'F')),
    CONSTRAINT check_tip_animal CHECK (UPPER(tip_animal) IN ('MAMIFER', 'PESTE', 'PASARE'))
);

---SUBTABEL MAMIFER
CREATE TABLE MAMIFER (
    id_animal NUMBER(5),
    perioada_gestatie NUMBER(3),
    tip_dieta VARCHAR2(30),
    CONSTRAINT pk_mamifer PRIMARY KEY (id_animal),
    CONSTRAINT fk_mamifer_animal FOREIGN KEY (id_animal) REFERENCES ANIMAL(id_animal) ON DELETE CASCADE,
    CONSTRAINT check_gestatie CHECK (perioada_gestatie>0),
    CONSTRAINT check_dieta_mamifer CHECK (UPPER(tip_dieta) IN ('ERBIVOR', 'CARNIVOR', 'OMNIVOR'))
);

---SUBTABEL PASARE
CREATE TABLE PASARE (
    id_animal NUMBER(5),
    anvergura_aripi NUMBER(5, 2), 
    este_zburatoare CHAR(2),
    CONSTRAINT pk_pasare PRIMARY KEY (id_animal),
    CONSTRAINT fk_pasare_animal FOREIGN KEY (id_animal) REFERENCES ANIMAL(id_animal) ON DELETE CASCADE,
    CONSTRAINT check_zburatoare CHECK (UPPER(este_zburatoare) IN ('DA', 'NU')),
    CONSTRAINT check_anvergura CHECK (anvergura_aripi>0)
);

---SUBTABEL PESTE
CREATE TABLE PESTE (
    id_animal NUMBER(5),
    tip_apa VARCHAR2(25),
    temp_apa NUMBER(4, 1),
    adancime_minima NUMBER(5, 2), -- în metri
    CONSTRAINT pk_peste PRIMARY KEY (id_animal),
    CONSTRAINT fk_peste_animal FOREIGN KEY (id_animal) REFERENCES ANIMAL(id_animal) ON DELETE CASCADE,
    CONSTRAINT check_tip_apa CHECK (UPPER(tip_apa) IN ('DULCE', 'SARATA')),
    CONSTRAINT check_adancime CHECK (adancime_minima>=0)
);

---TABEL HRANA
CREATE TABLE HRANA (
    id_hrana NUMBER(5),
    denumire VARCHAR2(50) NOT NULL,
    stoc_kg NUMBER(10, 2) DEFAULT 0,
    data_expirarii DATE,
    categorie VARCHAR2(30), 
    CONSTRAINT pk_hrana PRIMARY KEY (id_hrana),
    CONSTRAINT check_stoc CHECK (stoc_kg>=0),
    CONSTRAINT check_pret CHECK (pret_per_kg>0)
);

---TABEL FURNIZOR
CREATE TABLE FURNIZOR (
    id_furnizor NUMBER(5),
    nume VARCHAR2(100) NOT NULL,
    telefon CHAR(10),
    email VARCHAR2(100),
    oras VARCHAR2(50),
    cui VARCHAR2(15) NOT NULL,
    CONSTRAINT pk_furnizor PRIMARY KEY (id_furnizor),
    CONSTRAINT uk_cui UNIQUE (cui),  
    CONSTRAINT uk_email_furnizor UNIQUE (email)
);

---TABEL SPECTACOL
CREATE TABLE SPECTACOL (
    id_spectacol NUMBER(5),
    nume VARCHAR2(100) NOT NULL,
    locatie VARCHAR2(50) NOT NULL,
    data_spectacol DATE NOT NULL,
    durata_min NUMBER(3),
    cap_spectatori NUMBER(4),
    CONSTRAINT pk_spectacol PRIMARY KEY (id_spectacol),
    CONSTRAINT check_durata CHECK (durata_min>0),
    CONSTRAINT check_capacitate CHECK (cap_spectatori>0)
);

---TABEL VIIZTATOR
CREATE TABLE VIZITATOR (
    id_vizitator NUMBER(5),
    nume VARCHAR2(50) NOT NULL,
    prenume VARCHAR2(50) NOT NULL,
    email VARCHAR2(100),
    telefon CHAR(10),
    CONSTRAINT pk_vizitator PRIMARY KEY (id_vizitator),
    CONSTRAINT uk_email_vizitator UNIQUE (email),
    CONSTRAINT uk_tel_vizitator UNIQUE (telefon)
);


---TABEL BILET
CREATE TABLE BILET (
    id_bilet NUMBER(5),
    id_vizitator NUMBER(5) NOT NULL,
    tip_bilet VARCHAR2(20),
    pret NUMBER(6, 2) NOT NULL,
    data_valabilitate DATE NOT NULL,
    status VARCHAR2(20),
    CONSTRAINT pk_bilet PRIMARY KEY (id_bilet),
    CONSTRAINT fk_bilet_vizitator FOREIGN KEY (id_vizitator) 
        REFERENCES VIZITATOR(id_vizitator) ON DELETE CASCADE,
    CONSTRAINT check_pret_bilet CHECK (pret>=0),
    CONSTRAINT check_tip_bilet CHECK (UPPER(tip_bilet) IN ('ADULT', 'COPIL', 'STUDENT', 'PENSIONAR', 'GRUP')),
    CONSTRAINT check_status_bilet CHECK (UPPER(status) IN ('VALID', 'UTILIZAT', 'ANULAT', 'EXPIRAT'))
);

---TABEL RECENZIE
CREATE TABLE RECENZIE (
    id_recenzie NUMBER(5),
    id_vizitator NUMBER(5) NOT NULL,
    id_sector NUMBER(5) NOT NULL,
    nota NUMBER(1) NOT NULL,
    comentariu VARCHAR2(500),
    data_review DATE DEFAULT SYSDATE,
    CONSTRAINT pk_recenzie PRIMARY KEY (id_recenzie),
    CONSTRAINT fk_recenzie_vizitator FOREIGN KEY (id_vizitator) 
    REFERENCES VIZITATOR(id_vizitator) ON DELETE CASCADE,
    CONSTRAINT fk_recenzie_sector FOREIGN KEY (id_sector) 
    REFERENCES SECTOR(id_sector) ON DELETE CASCADE,
    CONSTRAINT check_nota CHECK (nota BETWEEN 1 AND 5)
);

---TABEL CONSULTATIE
CREATE TABLE CONSULTATIE (
    id_consultatie NUMBER(5),
    id_medic NUMBER(5) NOT NULL,
    id_animal NUMBER(5) NOT NULL,
    data_consultatie DATE DEFAULT SYSDATE NOT NULL,
    diagnostic VARCHAR2(100),
    tratament VARCHAR2(100),
    greutate_masurata NUMBER(10, 2),
    observatii VARCHAR2(200),
    CONSTRAINT pk_consultatie PRIMARY KEY (id_consultatie),
    CONSTRAINT fk_cons_medic FOREIGN KEY (id_medic) REFERENCES ANGAJAT(id_angajat),
    CONSTRAINT fk_cons_animal FOREIGN KEY (id_animal) REFERENCES ANIMAL(id_animal),
    CONSTRAINT check_greutate CHECK (greutate_masurata>0)
);
---TABEL ASOCIATIV ANIMAL_impreuna_cu_ANGAJAT_participa_la_SPECTACOL
CREATE TABLE PARTICIPA_LA (
    id_animal NUMBER(5),
    id_dresor NUMBER(5),
    id_spectacol NUMBER(5),
    ore_pregatire NUMBER(4, 1),
    bonus_financiar NUMBER(8, 2),
    stare_animal_final VARCHAR2(100),
    CONSTRAINT pk_participa_la PRIMARY KEY (id_animal, id_dresor, id_spectacol),
    CONSTRAINT fk_part_animal FOREIGN KEY (id_animal) REFERENCES ANIMAL(id_animal),
    CONSTRAINT fk_part_dresor FOREIGN KEY (id_dresor) REFERENCES ANGAJAT(id_angajat),
    CONSTRAINT fk_part_spectacol FOREIGN KEY (id_spectacol) REFERENCES SPECTACOL(id_spectacol)
);

---TABEL ASOCIATIV ANGAJAT_ingrijeste_ANIMAL
CREATE TABLE INGRIJESTE (
    id_animal NUMBER(5),
    id_ingrijitor NUMBER(5),
    ore_pe_sap NUMBER(3),
    tip_activitate VARCHAR2(50),
    data_inceput DATE DEFAULT SYSDATE,
    CONSTRAINT pk_ingrijeste PRIMARY KEY (id_animal, id_ingrijitor),
    CONSTRAINT fk_ingr_animal FOREIGN KEY (id_animal) REFERENCES ANIMAL(id_animal),
    CONSTRAINT fk_ingr_angajat FOREIGN KEY (id_ingrijitor) REFERENCES ANGAJAT(id_angajat)
);

---TABEL ASOCIATIV ANIMAL_mananca_HRANA
CREATE TABLE MANANCA (
    id_animal NUMBER(5),
    id_hrana NUMBER(5),
    cant_zilnica_kg NUMBER(6, 2),
    portii_pe_zi NUMBER(2),
    observatii VARCHAR2(200),
    CONSTRAINT pk_mananca PRIMARY KEY (id_animal, id_hrana),
    CONSTRAINT fk_mananca_animal FOREIGN KEY (id_animal) REFERENCES ANIMAL(id_animal),
    CONSTRAINT fk_mananca_hrana FOREIGN KEY (id_hrana) REFERENCES HRANA(id_hrana),
    CONSTRAINT check_cantitate CHECK (cant_zilnica_kg > 0)
);

---TABEL ASOCIATIV FURNIZOR_furnizeaza_MANCARE
CREATE TABLE FURNIZEAZA (
    id_hrana NUMBER(5),
    id_furnizor NUMBER(5),
    pret_per_kg NUMBER(8, 2),
    timp_livrare_zile NUMBER(3),
    cant_minima NUMBER(10, 2),
    CONSTRAINT pk_furnizeaza PRIMARY KEY (id_hrana, id_furnizor),
    CONSTRAINT fk_furn_hrana FOREIGN KEY (id_hrana) REFERENCES HRANA(id_hrana),
    CONSTRAINT fk_furn_furnizor FOREIGN KEY (id_furnizor) REFERENCES FURNIZOR(id_furnizor),
    CONSTRAINT check_pret_furniza CHECK (pret_per_kg > 0)
);
